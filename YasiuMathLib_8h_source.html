248 template<typename T>
249 T FindMinimalRadiusForIntersectingTangentsOnArc( const T alfa, const T beta, const T symmetricWidth )
250 {
251 /*
252 * sin (alfa) * A - sin(beta) * b
253 * R = ----------------------------------
254 * cos(beta) - cos(alfa)
255 */
256 T angleDiff = fmod(abs(alfa - beta), 360);
257 assert(symmetricWidth > 0);
258 assert(angleDiff > 0);
259 assert(angleDiff != 180); // Only place when they don't meet in infinite space
260
261 T nominator = symmetricWidth * (sin(alfa) + sin(beta));
262 T denominator = cos(beta) - cos(alfa);
263 T result = abs(nominator / denominator);
264
265 return result;
266 }

287 template<typename T>
288 std::pair<T, T> FindMinimalRadiusForIntersectingTangentsOnArcAsymmetric(
289 const T alfa,
290 const T beta,
291 const T widthA,
292 const T widthB
293 )
294 {
295 std::pair<T, T> result;
296
297 T angleDiff = fmod(abs(alfa - beta), 360); /* Keep angle in 0-360 range */
298 assert(widthA > 0);
299 assert(widthB > 0);
300 assert(angleDiff > 0);
301 assert(angleDiff != 180);
302
303 /* X - formula case */
304 T nominator1 = sin(alfa) * widthA + sin(beta) * widthB;
305 T denominator1 = cos(alfa) - cos(beta);
306 T radius1 = abs(nominator1 / denominator1);
307
308 /* Y - formula case */
309 T nominator2 = -(cos(alfa) * widthA + cos(beta) * widthB);
310 T denominator2 = sin(alfa) - sin(beta);
311 T radius2 = abs(nominator2 / denominator2);
312
313 if ( widthA < widthB ) {
314 /* Pack longer radius to first */
315 if ( radius1 > radius2 ) {
316 result.first = radius1;
317 result.second = radius2;
318 }
319 else {
320 result.first = radius2;
321 result.second = radius1;
322 }
323 }
324 else {
325 /* Pack shorter radius to first */
326 if ( radius1 < radius2 ) {
327 result.first = radius1;
328 result.second = radius2;
329 }
330 else {
331 result.first = radius2;
332 result.second = radius1;
333 }
334 }
335 return result;
336 }
