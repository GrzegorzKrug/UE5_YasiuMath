91 * cos(B) = adjacent / hypotenuse -> dx = cos * hypotenuse

93

155 template<typename T>
156 T FindMinimalRadiusForIntersectingTangentsOnArc( const T alfa, const T beta, const T symmetricWidth )
157 {
158 /*
159 * sin (alfa) * A - sin(beta) * b
160 * R = ----------------------------------
161 * cos(beta) - cos(alfa)
162 */
163 T angleDiff = fmod(abs(alfa - beta), 360);
164 assert(symmetricWidth > 0);
165 assert(angleDiff > 0);
166 assert(angleDiff != 180); // Only place when they don't meet in infinite space
167
168 T nominator = symmetricWidth * (sin(alfa) + sin(beta));
169 T denominator = cos(beta) - cos(alfa);
170 T result = abs(nominator / denominator);
171
172 return result;
173 }

194 template<typename T>
195 std::pair<T, T> FindMinimalRadiusForIntersectingTangentsOnArcAsymmetric(
196 const T alfa,
197 const T beta,
198 const T widthA,
199 const T widthB
200 )
201 {
202 std::pair<T, T> result;
203
204 T angleDiff = fmod(abs(alfa - beta), 360); /* Keep angle in 0-360 range */
205 assert(widthA > 0);
206 assert(widthB > 0);
207 assert(angleDiff > 0);
208 assert(angleDiff != 180);
209
210 /* X - formula case */
211 T nominator1 = sin(alfa) * widthA + sin(beta) * widthB;
212 T denominator1 = cos(alfa) - cos(beta);
213 T radius1 = abs(nominator1 / denominator1);
214
215 /* Y - formula case */
216 T nominator2 = -(cos(alfa) * widthA + cos(beta) * widthB);
217 T denominator2 = sin(alfa) - sin(beta);
218 T radius2 = abs(nominator2 / denominator2);
219
220 if ( widthA < widthB ) {
221 /* Pack longer radius to first */
222 if ( radius1 > radius2 ) {
223 result.first = radius1;
224 result.second = radius2;
225 }
226 else {
227 result.first = radius2;
228 result.second = radius1;
229 }
230 }
231 else {
232 /* Pack shorter radius to first */
233 if ( radius1 < radius2 ) {
234 result.first = radius1;
235 result.second = radius2;
236 }
237 else {
238 result.first = radius2;
239 result.second = radius1;
240 }
241 }
242 return result;
243 }

622 template<typename T>
623 static T Remap( const T Value, const T InMin, const T InMax, T OutMin, T OutMax, const bool ClampOutput = false )
624 {
625 const T SubVal = Value - InMin;
626 const T Divid = InMax - InMin;
627
628 if ( OutMax < OutMin ) {
629 std::swap(OutMin, OutMax);
630 }
631
632 if ( Numeric::IsNearly0(Divid) ) {
633 /* What to return? Any result will not be accurate, but 0 division goes to max */
634 return OutMax;
635 }
636
637 const T Factor = SubVal / Divid;
638
639
640 const T SubValOut = OutMax - OutMin;
641 const T Output = (Factor * SubValOut) + OutMin;
642 if ( ClampOutput ) {
643 return std::clamp(Output, OutMin, OutMax);
644 }
645
646 return Output;
647 }
