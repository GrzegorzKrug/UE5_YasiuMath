167 template<typename T>
168 T FindMinimalRadiusForIntersectingTangentsOnArc( const T alfa, const T beta, const T symmetricWidth )
169 {
170 /*
171 * sin (alfa) * A - sin(beta) * b
172 * R = ----------------------------------
173 * cos(beta) - cos(alfa)
174 */
175 T angleDiff = fmod(abs(alfa - beta), 360);
176 assert(symmetricWidth > 0);
177 assert(angleDiff > 0);
178 assert(angleDiff != 180); // Only place when they don't meet in infinite space
179
180 T nominator = symmetricWidth * (sin(alfa) + sin(beta));
181 T denominator = cos(beta) - cos(alfa);
182 T result = abs(nominator / denominator);
183
184 return result;
185 }

206 template<typename T>
207 std::pair<T, T> FindMinimalRadiusForIntersectingTangentsOnArcAsymmetric(
208 const T alfa,
209 const T beta,
210 const T widthA,
211 const T widthB
212 )
213 {
214 std::pair<T, T> result;
215
216 T angleDiff = fmod(abs(alfa - beta), 360); /* Keep angle in 0-360 range */
217 assert(widthA > 0);
218 assert(widthB > 0);
219 assert(angleDiff > 0);
220 assert(angleDiff != 180);
221
222 /* X - formula case */
223 T nominator1 = sin(alfa) * widthA + sin(beta) * widthB;
224 T denominator1 = cos(alfa) - cos(beta);
225 T radius1 = abs(nominator1 / denominator1);
226
227 /* Y - formula case */
228 T nominator2 = -(cos(alfa) * widthA + cos(beta) * widthB);
229 T denominator2 = sin(alfa) - sin(beta);
230 T radius2 = abs(nominator2 / denominator2);
231
232 if ( widthA < widthB ) {
233 /* Pack longer radius to first */
234 if ( radius1 > radius2 ) {
235 result.first = radius1;
236 result.second = radius2;
237 }
238 else {
239 result.first = radius2;
240 result.second = radius1;
241 }
242 }
243 else {
244 /* Pack shorter radius to first */
245 if ( radius1 < radius2 ) {
246 result.first = radius1;
247 result.second = radius2;
248 }
249 else {
250 result.first = radius2;
251 result.second = radius1;
252 }
253 }
254 return result;
255 }

623 ProjectileDynamicState PredictQuick( T QueryTime ) const
624 {
625 /* Static because must return bool ( can not return Copied value )
626 * Formula:
627 * | V + At | = V_max
628 * V_max is scalar, max speed
629 * V - Velocity vec
630 * A - Acceleration/Thrust vec
631 * t - time
632 *
633 * 0 = t^2 * A^2 + 2t * VA + V^2 - V_max^2
634 * Now-> do components sum x+y+z
635 */
636 ProjectileDynamicState<T> ProjectileState = *this;
637
638 if ( ProjectileState.Acceleration.IsNearly0() ) {
639 ProjectileState.Position = ProjectileState.Position + ProjectileState.Velocity * QueryTime;
640 return ProjectileState;
641 }
642
643 const T Xa = ProjectileState.Acceleration.LengthSquared();
644 const T Xb = 2 * (ProjectileState.Velocity * ProjectileState.Acceleration).Sum();
645 const T Xc = ProjectileState.Velocity.LengthSquared() - ProjectileState.MaxSpeed * ProjectileState.MaxSpeed;
646
647 const T delta = Xb * Xb - 4 * Xa * Xc;
648 T TimeToMaxSpeed = -1;
649
650 if ( delta < 0 ) {
651 /* Technically should never occur
652 * Case(Accel=0) is computed before (Accel 0 can't fall into negative delta condition)
653 */
654 throw std::runtime_error("Delta negative. Can't return false. Solution is not Valid at all.");
655 // return false;
656 }
657 else if ( fabs(delta) <= YasiuMath::Constants::EPSILON ) {
658 /* 1 solution close to ~0 */
659 TimeToMaxSpeed = -Xb / (2 * Xa);
660 }
661 else {
662 /* 2 Solutions */
663 T x1 = (-Xb - std::sqrt(delta)) / (2 * Xa);
664 T x2 = (-Xb + std::sqrt(delta)) / (2 * Xa);
665
666 if ( x1 > 0 ) {
667 TimeToMaxSpeed = x1;
668 }
669 else if ( x2 > 0 ) {
670 TimeToMaxSpeed = x2;
671 }
672
673 if ( x1 > 0 && x2 > 0 && x1 < x2 ) {
674 /* Pick smaller X only when both are valid */
675 TimeToMaxSpeed = x1;
676 }
677 }
678
679 /*
680 * /\ Accel
681 * | / |
682 * | / |
683 * | / |
684 * | / X |
685 * ----------> Time
686 * X - Area is velocity -> Triangle formula
687 */
688 /* TimeMax<0 to catch error when limit is set 0, not to use <= to not use clamping */
689 if ( QueryTime < TimeToMaxSpeed || TimeToMaxSpeed < 0 ) {
690 /* Travel to MaxSpeed only */
691 ProjectileState.Position += ProjectileState.Velocity * QueryTime + ProjectileState.Acceleration * (0.5 *
692 QueryTime * QueryTime);
693 ProjectileState.Velocity += ProjectileState.Acceleration * QueryTime;
694 }
695 else {
696 /* Acceleration Phase */
697 ProjectileState.Position += ProjectileState.Velocity * TimeToMaxSpeed + ProjectileState.Acceleration * (0.5 *
698 TimeToMaxSpeed * TimeToMaxSpeed);
699 /* Account for velocity changes */
700 ProjectileState.Velocity += ProjectileState.Acceleration * TimeToMaxSpeed;
701
702 /* Normalize to Max */
703 ProjectileState.Velocity = ProjectileState.Velocity.NormalizeInPlace() * ProjectileState.MaxSpeed;
704 /* Max Speed reached */
705 float RestofTime = QueryTime - TimeToMaxSpeed;
706 if ( RestofTime > 0 ) {
707 /* Add rest travel time */
708 ProjectileState.Position += ProjectileState.Velocity * RestofTime;
709 }
710 }
711
712 return ProjectileState;
713 }

724 virtual void DiscreteStep( T deltaTime )
725 {
726 auto oldVel = Velocity;
727 /* Acceleration */
728 if ( !Acceleration.IsNearly0() ) {
729 Velocity += (Acceleration) * deltaTime;
730 }
731
732 /* Friction as last step to damp all forces */
733 if ( AirFrictionCoeff > 0 ) {
734 /* Preserve sign! not loose it with square */
735 Vec3<T> drag = (Velocity * Velocity.Abs()) * AirFrictionCoeff * deltaTime;
736 if ( drag.Length() >= Velocity.Length() ) {
737 Velocity = 0;
738 }
739 else {
740 Velocity -= drag;
741 }
742 }
743
744
745 /* Speed limit */
746 /* Not need for checking Velocity!=0 safety/optimization */
747 if ( MaxSpeed > 0 && Velocity.Length() > MaxSpeed ) {
748 Velocity = Velocity.NormalizeInPlace() * MaxSpeed;
749 }
750
751 Position += (oldVel + Velocity) * (deltaTime * 0.5);
752 };

754
767 void AutoStep( T QueryTime, T DeltaTime )
768 {
769 if ( AirFrictionCoeff == 0 ) {
770 const auto Result = PredictQuick(QueryTime);
771 *this = Result;
772 }
773 else {
774 const int N = QueryTime / DeltaTime;
775 for ( int i = 0; i < N; i++ ) {
776 DiscreteStep(DeltaTime);
777 }
778
779 const T missingTime = QueryTime - (DeltaTime * N);
780 if ( missingTime > 0 ) {
781 DiscreteStep(missingTime);
782 }
783 }

876 template<typename T>
877 T Remap( const T Value, const T InMin, const T InMax, T OutMin, T OutMax, const bool ClampOutput = false )
878 {
879 const T SubVal = Value - InMin;
880 const T Divid = InMax - InMin;
881
882 if ( OutMax < OutMin ) {
883 std::swap(OutMin, OutMax);
884 }
885
886 if ( Numeric::IsNearly0(Divid) ) {
887 /* What to return? Any result will not be accurate, but 0 division goes to max */
888 return OutMax;
889 }
890
891 const T Factor = SubVal / Divid;
892
893
894 const T SubValOut = OutMax - OutMin;
895 const T Output = (Factor * SubValOut) + OutMin;
896 if ( ClampOutput ) {
897 return std::clamp(Output, OutMin, OutMax);
898 }
899
900 return Output;
901 }
