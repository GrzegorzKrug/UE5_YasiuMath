380 virtual void MultiStep( T QueryTime, T DeltaTime )
381 {
382 const int N = QueryTime / DeltaTime;
383 for ( int i = 0; i < N; i++ ) {
384 Step(DeltaTime);
385 }
386
387 T missingTime = QueryTime - (DeltaTime * N);
388 if ( missingTime > EPSILON ) {
389 Step(missingTime);
390 }
391 }

397 virtual void Step( T deltaTime )
398 {
399 /* Acceleration */
400 if ( !ThrustVector.IsNearly0() ) {
401 Velocity += ThrustVector * deltaTime;
402 }
403
404 /* Friction as last step to damp all forces */
405 if ( AirFrictionCoeff > 0 ) {
406 /* Preserve sign! not loose it with square */
407 Vec3<T> drag = (Velocity * Velocity.Abs()) * AirFrictionCoeff * deltaTime;
408 Velocity -= drag;
409 }
410
411
412 /* Speed limit */
413 // if ( !ThrustVector.IsNearly0() ) {
414 /* Not need for checking Velocity!=0 safety/optimization */
415 if ( Velocity.Length() > MaxSpeed ) {
416 Velocity = Velocity.Normalize() * MaxSpeed;
417 }
418 // }
419
420 Position += Velocity * deltaTime;
421 };

436 static bool PredictQuick( ProjectileDynamicState& ProjectileState, T QueryTime )
437 {
438 /* Static because must return bool ( can not return Copied value )
439 * Formula:
440 * | V + At | = V_max
441 * V_max is scalar, max speed
442 * V - Velocity vec
443 * A - Acceleration/thrust vec
444 * t - time
445 *
446 * 0 = t^2*A^2 + 2t*VA + V^2-V_max^2
447 * Now-> do components sum x+y+z
448 */
449 // PredictionState = *this;
450
451 // OutState = *this;
452 if ( ProjectileState.ThrustVector.IsNearly0() ) {
453 ProjectileState.Position = ProjectileState.Position + ProjectileState.Velocity * QueryTime;
454 return true;
455 }
456
457 T Xa = ProjectileState.ThrustVector.LengthSquared();
458 T Xb = 2 * (ProjectileState.Velocity * ProjectileState.ThrustVector).Sum();
459 T Xc = ProjectileState.Velocity.LengthSquared() - ProjectileState.MaxSpeed * ProjectileState.MaxSpeed;
460
461 T delta = Xb * Xb - 4 * Xa * Xc;
462 T TimeToMaxSpeed = -1;
463 if ( delta < 0 ) {
464 /* Technically should never occur
465 * Accel=0 is computed before
466 */
467 throw std::runtime_error("Delta negative. Can't return false. Solution is not Valid at all.");
468 // return false;
469 }
470 else if ( fabs(delta) <= EPSILON ) {
471 /* 1 solution ~ 0 */
472 TimeToMaxSpeed = -Xb / (2 * Xa);
473 }
474 else {
475 /* 2 Solutions */
476 T x1 = (-Xb - std::sqrt(delta)) / (2 * Xa);
477 T x2 = (-Xb + std::sqrt(delta)) / (2 * Xa);
478
479 if ( x1 > 0 ) {
480 TimeToMaxSpeed = x1;
481 }
482 else if ( x2 > 0 ) {
483 TimeToMaxSpeed = x2;
484 }
485 if ( x1 > 0 && x2 > 0 && x1 < x2 ) {
486 TimeToMaxSpeed = x1;
487 }
488 }
489
490 if ( TimeToMaxSpeed < 0 ) {
491 /* Wrong params? */
492 return false;
493 }
494 if ( QueryTime <= TimeToMaxSpeed ) {
495 return true;
496 }
497
498
499 /* Estimated, No Speed Clipping */
500 return false;
501 }

511 ProjectileDynamicState Predict( T QueryTime, T DeltaTime ) const
512 {
513 ProjectileDynamicState object = *this;
514 object.MultiStep(QueryTime, DeltaTime);
515 return object;
516 }
