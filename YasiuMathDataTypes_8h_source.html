377 virtual void MultiStep( T QueryTime, T DeltaTime )
378 {
379 const int N = QueryTime / DeltaTime;
380 for ( int i = 0; i < N; i++ ) {
381 Step(DeltaTime);
382 }
383
384 T missingTime = QueryTime - (DeltaTime * N);
385 if ( missingTime > YasiuMath::Constants::EPSILON ) {
386 Step(missingTime);
387 }
388 }
389
394 virtual void Step( T deltaTime )
395 {
396 /* Acceleration */
397 if ( !ThrustVector.IsNearly0() ) {
398 Velocity += ThrustVector * deltaTime;
399 }
400
401 /* Friction as last step to damp all forces */
402 if ( AirFrictionCoeff > 0 ) {
403 /* Preserve sign! not loose it with square */
404 Vec3<T> drag = (Velocity * Velocity.Abs()) * AirFrictionCoeff * deltaTime;
405 Velocity -= drag;
406 }
407
408
409 /* Speed limit */
410 // if ( !ThrustVector.IsNearly0() ) {
411 /* Not need for checking Velocity!=0 safety/optimization */
412 if ( Velocity.Length() > MaxSpeed ) {
413 Velocity = Velocity.Normalize() * MaxSpeed;
414 }
415 // }
416
417 Position += Velocity * deltaTime;
418 };

433 static bool PredictQuick( ProjectileDynamicState& ProjectileState, T QueryTime )
434 {
435 /* Static because must return bool ( can not return Copied value )
436 * Formula:
437 * | V + At | = V_max
438 * V_max is scalar, max speed
439 * V - Velocity vec
440 * A - Acceleration/thrust vec
441 * t - time
442 *
443 * 0 = t^2*A^2 + 2t*VA + V^2-V_max^2
444 * Now-> do components sum x+y+z
445 */
446 // PredictionState = *this;
447
448 // OutState = *this;
449 if ( ProjectileState.ThrustVector.IsNearly0() ) {
450 ProjectileState.Position = ProjectileState.Position + ProjectileState.Velocity * QueryTime;
451 return true;
452 }
453
454 T Xa = ProjectileState.ThrustVector.LengthSquared();
455 T Xb = 2 * (ProjectileState.Velocity * ProjectileState.ThrustVector).Sum();
456 T Xc = ProjectileState.Velocity.LengthSquared() - ProjectileState.MaxSpeed * ProjectileState.MaxSpeed;
457
458 T delta = Xb * Xb - 4 * Xa * Xc;
459 T TimeToMaxSpeed = -1;
460 if ( delta < 0 ) {
461 /* Technically should never occur
462 * Accel=0 is computed before
463 */
464 throw std::runtime_error("Delta negative. Can't return false. Solution is not Valid at all.");
465 // return false;
466 }
467 else if ( fabs(delta) <= YasiuMath::Constants::EPSILON ) {
468 /* 1 solution ~ 0 */
469 TimeToMaxSpeed = -Xb / (2 * Xa);
470 }
471 else {
472 /* 2 Solutions */
473 T x1 = (-Xb - std::sqrt(delta)) / (2 * Xa);
474 T x2 = (-Xb + std::sqrt(delta)) / (2 * Xa);
475
476 if ( x1 > 0 ) {
477 TimeToMaxSpeed = x1;
478 }
479 else if ( x2 > 0 ) {
480 TimeToMaxSpeed = x2;
481 }
482 if ( x1 > 0 && x2 > 0 && x1 < x2 ) {
483 TimeToMaxSpeed = x1;
484 }
485 }
486
487 if ( TimeToMaxSpeed < 0 ) {
488 /* Wrong params? */
489 return false;
490 }
491 if ( QueryTime <= TimeToMaxSpeed ) {
492 return true;
493 }
494
495
496 /* Estimated, No Speed Clipping */
497 return false;
498 }
